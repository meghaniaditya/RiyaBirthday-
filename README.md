    function braid(g, cx, cy, rx, ry, count) {
      for (var i = 0; i <= count; i++) {
        var th = Math.PI * i / count;
        el("ellipse", {
          cx: cx + rx * Math.cos(th), cy: cy + ry * Math.sin(th) - 4, rx: 9, ry: 4.6,
          transform: "rotate(35 " + (cx + rx * Math.cos(th)) + " " + (cy + ry * Math.sin(th) - 4) + ")",
          fill: "#ffffff", stroke: "#f0a9c2", "stroke-width": 1.4
        }, g);
      }
    }
    braid($("braidBottom"), 320, 540, 200, 34, 30);
    braid($("braidTop"), 320, 430, 130, 28, 20);

    // drip icing around the top tier
    var drips = $("drips");
    el("path", { d: "M190,330 A130,28 0 0 0 450,330", fill: "none", stroke: "#f08fb1", "stroke-width": 13, "stroke-linecap": "round" }, drips);
    var lens = [18, 28, 14, 26, 20, 30, 16, 28, 22, 30];

    }

    var g = $("candles"), N = 19, cols = ["#ffffff", "#ff9ec0", "#ffe08a", "#cdb8ff"], list = [];
    for (var n = 0; n < N; n++) {
      var th = (2 * Math.PI * n) / N - Math.PI / 2;
      list.push({ n: n, x: 320 + 104 * Math.cos(th), y: 330 + 17 * Math.sin(th) });
    }
    list.sort(function (a, b) { return a.y - b.y; });
    list.forEach(function (c) {
      var cg = el("g", { class: "candle" }, g);

  function rain() {
    if (reduceMotion) return;
    for (var i = 0; i < 120; i++) {
      ps.push({
        x: Math.random() * innerWidth, y: -20 - Math.random() * innerHeight * 0.6,
        vy: 1.2 + Math.random() * 1.8, sway: Math.random() * 2 * Math.PI, sv: 0.02 + Math.random() * 0.03,
        amp: 0.8 + Math.random() * 1.6, r: Math.random() * Math.PI, vr: (Math.random() - 0.5) * 0.06,
        w: 5 + Math.random() * 5, h: 10 + Math.random() * 8, c: PCOLS[(Math.random() * PCOLS.length) | 0]
      });
    }
    if (!raf) raf = requestAnimationFrame(petalFrame);
  }
  function petalFrame() {
    cx.clearRect(0, 0, innerWidth, innerHeight);
    for (var i = ps.length - 1; i >= 0; i--) {
      var p = ps[i];
      p.sway += p.sv; p.x += Math.sin(p.sway) * p.amp; p.y += p.vy; p.r += p.vr;
      cx.save(); cx.translate(p.x, p.y); cx.rotate(p.r); cx.fillStyle = p.c;
      cx.beginPath(); cx.ellipse(0, 0, p.w / 2, p.h / 2, 0, 0, 2 * Math.PI); cx.fill(); cx.restore();
      if (p.y > innerHeight + 30) ps.splice(i, 1);
    }
    if (ps.length) raf = requestAnimationFrame(petalFrame);
    else { raf = null; cx.clearRect(0, 0, innerWidth, innerHeight); }
  }

  /* ---------- petal geometry (curved, cupped, with a ruffled edge) ---------- */
  function petalGeo(o) {
    var J = o.segV || 22, I = o.segU || 8, j, i;
    var cl = [{ x: 0, y: 0, th: o.th0 }];
    for (j = 1; j <= J; j++) {
      var v = j / J, th = o.th0 + o.th1 * Math.pow(v, o.pw || 1.6), p = cl[j - 1];
      cl.push({ x: p.x + o.L * Math.sin(th) / J, y: p.y + o.L * Math.cos(th) / J, th: th });
    }
    var pos = [], uv = [], idx = [];
    for (j = 0; j <= J; j++) {
      var vv = j / J, hw = o.W * 0.5 * Math.sin(Math.PI * Math.pow(vv, o.wp || 0.78)), c = cl[j];
      var nx = Math.cos(c.th), ny = -Math.sin(c.th);
      for (i = 0; i <= I; i++) {
        var u = i / I, e = Math.abs(u - 0.5) * 2, z = (u - 0.5) * 2 * hw;
        var d = -(o.cup || 0) * hw * e * e + (o.rip || 0) * Math.sin(vv * 15 + (o.ph || 0)) * e * hw * vv;
        pos.push(c.x + nx * d, c.y + ny * d, z);
        uv.push(u, vv);
      }
    }
    for (j = 0; j < J; j++) for (i = 0; i < I; i++) {
      var a = j * (I + 1) + i, b = a + I + 1;
      idx.push(a, b, a + 1, b, b + 1, a + 1);
    }
    var g = new T.BufferGeometry();
    g.setAttribute("position", new T.Float32BufferAttribute(pos, 3));
    g.setAttribute("uv", new T.Float32BufferAttribute(uv, 2));
    g.setIndex(idx);
    g.computeVertexNormals();
    return g;
  }
  // open shape + a closed "bud" shape as a morph target, so petals really blossom
  function morphPetal(open, closed) {
    var g = petalGeo(open), c = petalGeo(closed);
    g.morphAttributes.position = [c.getAttribute("position")];
    g.morphAttributes.normal = [c.getAttribute("normal")];
    return g;
  }
  var geoCache = {};
  function petalGeometry(outer) {
    var k = outer ? "o" : "i";
    if (!geoCache[k]) {
      geoCache[k] = outer

    }
    return geoCache[k];
  }

    var root = new T.Group(), spin = new T.Group();

    var petals = [];
    for (var i = 0; i < 6; i++) {
      var outer = i % 2 === 0;

      var pivot = new T.Group(); pivot.rotation.y = i * Math.PI / 3; pivot.add(hinge);
      spin.add(pivot);
      petals.push({ mesh: mesh, delay: (outer ? 0 : 0.12) + (i % 3) * 0.05 });
    }
    var ctr = new T.Group(); spin.add(ctr);
    var filMat = new T.MeshStandardMaterial({ color: 0xf4ead6, roughness: 0.6 });
    var antMat = new T.MeshStandardMaterial({ color: 0xe07a1f, roughness: 0.55, emissive: 0x4a1a00 });
    var Yv = new T.Vector3(0, 1, 0);
    for (var k = 0; k < 6; k++) {
      var a = k * Math.PI / 3 + Math.PI / 6, ca = Math.cos(a), sa = Math.sin(a);
      var curve = new T.QuadraticBezierCurve3(new T.Vector3(0, 0.05, 0), new T.Vector3(ca * 0.25, 1.0, sa * 0.25), new T.Vector3(ca * 0.85, 1.85, sa * 0.85));
      ctr.add(new T.Mesh(new T.TubeGeometry(curve, 10, 0.022, 6, false), filMat));
      var ant = new T.Mesh(new T.SphereGeometry(1, 10, 8), antMat);
      ant.scale.set(0.07, 0.22, 0.07);
      ant.position.copy(curve.getPoint(1));
      var dir = curve.getTangent(1).clone().add(new T.Vector3(ca * 0.9, 0, sa * 0.9)).normalize();
      ant.quaternion.setFromUnitVectors(Yv, dir);
      ctr.add(ant);
    }
    var pc = new T.QuadraticBezierCurve3(new T.Vector3(0, 0, 0), new T.Vector3(0.05, 1.1, 0), new T.Vector3(0.15, 2.1, 0.05));

  }

      if (!dragging) { vel += (auto - vel) * Math.min(1, dt * 1.6); }
      rot += vel * dt;
    }
    ring.rotation.y = rot;

    gd += ((focus ? 1 : 0) - gd) * Math.min(1, dt * 6);
    cards.forEach(function (c) {
      var target = c === focus ? 1 : 0;
      if (reduce) c.f = target; else c.f += Math.max(-dt * 2.4, Math.min(dt * 2.4, target - c.f));
      var e = ease(c.f);
      var fy = reduce ? 0 : Math.sin(t * 0.8 + c.phase) * 0.12;
      c.group.position.set(c.slotPos.x, fy, c.slotPos.z);
      c.group.quaternion.copy(c.slotQuat);
      if (e > 0) {
        c.group.position.lerp(tgt.pos, e);
        c.group.quaternion.slerp(tgt.quat, e);
      }
      c.group.rotation.z += reduce ? 0 : Math.sin(t * 0.6 + c.phase) * 0.03 * (1 - e);
      c.group.scale.setScalar(1 + 0.32 * e);
      var dim = c === focus ? 1 : 1 - 0.62 * gd;
      c.frameMat.color.setScalar(dim); c.photoMat.color.setScalar(dim); c.capMat.color.setScalar(dim);
      c.tapeMat.opacity = 0.7 * (c === focus ? 1 : 1 - 0.5 * gd);
    });

    // drifting petals
    var arr = pgeo.attributes.position.array;
    for (var i = 0; i < PN; i++) {
      if (!reduce) { arr[i * 3 + 1] += pseed[i].sp * dt * 0.8; arr[i * 3] += Math.sin(t * 0.5 + pseed[i].ph) * dt * 0.15; }
      if (arr[i * 3 + 1] > 6.5) arr[i * 3 + 1] = -2;
    }
    pgeo.attributes.position.needsUpdate = true;

    px += (tx - px) * 0.05; py += (ty - py) * 0.05;
    camera.position.set(px * 0.9, 2.6 + py * 0.6, camZ);
    camera.lookAt(0, 0.3, 0);
  }
  function renderOnce() { update(performance.now() / 1000, 0.016); renderer.render(scene, camera); }
  function fail() {
    running = false;
    sec.classList.remove("photos-3d"); box.hidden = true;
    try { renderer.dispose(); } catch (e) {}
    if (dom.parentNode) dom.parentNode.removeChild(dom);
  }
  function frame(now) {
    if (!running) return;
    requestAnimationFrame(frame);
    try {
      lastDt = Math.min(0.05, (now - last) / 1000 || 0.016); last = now;
      update(now / 1000, lastDt);
      renderer.render(scene, camera);
    } catch (e) { fail(); }
  }

  if ("IntersectionObserver" in window) {
    new IntersectionObserver(function (en) {
      var vis = en[0].isIntersecting;
      if (vis) size();
      if (vis && !running) { running = true; last = performance.now(); requestAnimationFrame(frame); }
      else if (!vis) { running = false; }
    }, { threshold: 0.05 }).observe(mount);
  } else { running = true; requestAnimationFrame(frame); }
  if ("ResizeObserver" in window) new ResizeObserver(function () { size(); if (!running) { try { renderOnce(); } catch (e) {} } }).observe(mount);
  window.addEventListener("resize", size);

  try {
    window.lilyGallery = { set: setItems };
    setItems(window.lilyPhotoItems || []);
    sec.classList.add("photos-3d");
    box.hidden = false;
  } catch (e) { fail(); }
})();
